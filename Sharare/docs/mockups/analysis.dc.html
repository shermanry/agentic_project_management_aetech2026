def flight_window(weather, permit, limits):
    days = []
    for day in permit.authorized_days:
        ok = [h for h in weather.hours(day, 7, 17)
              if h.gust_mph < limits.max_gust_mph
              and h.rain_prob < limits.max_rain_prob
              and h.visibility_mi > limits.min_visibility_mi]
        run = longest_consecutive(ok)
        if len(run) >= limits.min_consecutive_hours:
            days.append({"day": day, "start": run[0], "end": run[-1]})
    return days
